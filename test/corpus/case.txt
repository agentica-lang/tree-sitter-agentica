def main() -> Void
  case shape do
    Shape.Circle(r) -> puts(r)
    Shape.Rect(w, _) ->
      puts(w)
      puts("rect")
    M.Shape.Empty -> puts("empty")
    1 -> 1
    -2 -> 2
    "s" -> 3
    true -> 4
    nil -> 5
    ok(v) -> 6
    error(_) -> 7
    _ -> puts("other")
  end
end
